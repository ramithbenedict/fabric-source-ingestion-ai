## Example 1 — CREATE TABLE with Identity, Precision, Hash, Constraints

### Input

Instruction: "Create a table."

### Output

```sql
CREATE TABLE [DBO].[DIM_PROJECT] (
    [PROJECT_ID]             BIGINT IDENTITY
    , [PROJECT_CODE]         VARCHAR(50)     NOT NULL
    , [PROJECT_NAME]         VARCHAR(200)    NOT NULL
    , [PROJECT_START_DATE]   DATE
    , [PROJECT_UPDATED_TS]   DATETIME2(6)
    , [PROJECT_LOAD_TIME]    TIME(0)
    , [PROJECT_ROW_HASH]     VARBINARY(32)
    , [PROJECT_ACTIVE_FLAG]  BIT
)
;
GO

ALTER TABLE [DBO].[DIM_PROJECT]
    ADD CONSTRAINT [PK_DIM_PROJECT] PRIMARY KEY NONCLUSTERED ([PROJECT_ID]) NOT ENFORCED
;
GO

ALTER TABLE [DBO].[DIM_PROJECT]
    ADD CONSTRAINT [UQ_PROJECT_CODE] UNIQUE NONCLUSTERED ([PROJECT_CODE]) NOT ENFORCED
;
GO

CREATE TABLE [DBO].[FACT_COST] (
    [COST_ID]            BIGINT IDENTITY
    , [PROJECT_FK]       BIGINT          NOT NULL
    , [COST_DATE]        DATE            NOT NULL
    , [COST_AMOUNT]      DECIMAL(19, 4)
    , [CURRENCY_CODE]    VARCHAR(3)
    , [COST_LOADED_TS]   DATETIME2(6)
)
;
GO

ALTER TABLE [DBO].[FACT_COST]
    ADD CONSTRAINT [PK_FACT_COST] PRIMARY KEY NONCLUSTERED ([COST_ID]) NOT ENFORCED
;
GO

ALTER TABLE [DBO].[FACT_COST]
    ADD CONSTRAINT [FK_FACT_COST_PROJECT] FOREIGN KEY ([PROJECT_FK])
    REFERENCES [DBO].[DIM_PROJECT] ([PROJECT_ID]) NOT ENFORCED
;
GO
```
